```bash
# cat test_perf.sh
while true
do
aa=`date +%T | awk -F: '{print $1":"$2}'`
bb=`date +%T | awk -F: '{print $3}'`

echo ----${aa}---${bb}---
if [[ ${aa} == "15:18" ]] && [[ ${bb} -ge 24 ]] && [[ ${bb} -le 26 ]]; then
    perf record -g -a -e skb:kfree_skb -o my_perf_data.data  -- sleep 3
else
    echo ----not ok--
    sleep 0.5
fi

done
```
